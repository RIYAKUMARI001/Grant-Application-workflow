┌─────────────────────────────────────────────────────────────────────────┐
│ 1. USER MODEL (core/models.py)                                         │
├─────────────────────────────────────────────────────────────────────────┤
│                                                                         │
│   class User(AbstractUser):                                            │
│       ROLE_CHOICES = [                                                 │
│           ('applicant', 'Applicant'),                                  │
│           ('reviewer', 'Reviewer'),                                    │
│           ('admin', 'Admin'),                                          │
│       ]                                                                │
│       role = models.CharField(choices=ROLE_CHOICES)                    │
│                                                                         │
│       def is_applicant(self): return self.role == 'applicant'          │
│       def is_reviewer(self): return self.role == 'reviewer'            │
│       def is_admin_role(self): return self.role == 'admin'             │
│                                                                         │
└─────────────────────────────────────────────────────────────────────────┘

┌─────────────────────────────────────────────────────────────────────────┐
│ 2. REGISTRATION FORM (core/forms.py)                                   │
├─────────────────────────────────────────────────────────────────────────┤
│                                                                         │
│   class UserRegistrationForm(UserCreationForm):                        │
│       email = forms.EmailField(required=True)                          │
│       role = forms.ChoiceField(choices=User.ROLE_CHOICES)              │
│                                                                         │
│       class Meta:                                                      │
│           model = User                                                 │
│           fields = ['username', 'email', 'password1',                  │
│                     'password2', 'role']                               │
│                                                                         │
└─────────────────────────────────────────────────────────────────────────┘

┌─────────────────────────────────────────────────────────────────────────┐
│ 3. REGISTRATION VIEW (core/views.py)                                   │
├─────────────────────────────────────────────────────────────────────────┤
│                                                                         │
│   def register_view(request):                                          │
│       if request.method == 'POST':                                     │
│           form = UserRegistrationForm(request.POST)                    │
│           if form.is_valid():                                          │
│               user = form.save()                                       │
│               login(request, user)  # Auto-login                       │
│                                                                         │
│               # Role-based redirect                                    │
│               if user.is_applicant():                                  │
│                   return redirect('applicant_dashboard')               │
│               elif user.is_reviewer():                                 │
│                   return redirect('reviewer_dashboard')                │
│               elif user.is_admin_role():                               │
│                   return redirect('admin_dashboard')                   │
│                                                                         │
└─────────────────────────────────────────────────────────────────────────┘

┌─────────────────────────────────────────────────────────────────────────┐
│ 4. LOGIN VIEW (core/views.py)                                          │
├─────────────────────────────────────────────────────────────────────────┤
│                                                                         │
│   class CustomLoginView(LoginView):                                    │
│       template_name = 'registration/login.html'                        │
│                                                                         │
│       def get_success_url(self):                                       │
│           user = self.request.user                                     │
│           if user.is_applicant():                                      │
│               return '/applicant/dashboard/'                           │
│           elif user.is_reviewer():                                     │
│               return '/reviewer/dashboard/'                            │
│           elif user.is_admin_role():                                   │
│               return '/admin-panel/dashboard/'                         │
│                                                                         │
└─────────────────────────────────────────────────────────────────────────┘

                            URL ROUTING

═══════════════════════════════════════════════════════════════════════════

    /                           →  home (landing page)
    /register/                  →  register_view
    /login/                     →  CustomLoginView
    /logout/                    →  logout_view
    /applicant/dashboard/       →  applicant_dashboard
    /reviewer/dashboard/        →  reviewer_dashboard
    /admin-panel/dashboard/     →  admin_dashboard
